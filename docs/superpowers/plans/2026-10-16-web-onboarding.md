# Web Onboarding Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** After `docker compose up -d`, a homelab operator finishes the whole first install in the browser: first admin, server settings, invites, the EDICT2/Conjugations dictionary, and a desktop `ta.db` import.

**Architecture:** An unauthenticated `/setup` API creates the first admin under a Postgres advisory lock. A new `server_settings` table overrides five env settings (DB row > env > default) through one lookup, `api.server_settings.effective(db)`. Dictionary files and `ta.db` uploads live under a new `TA_DATA_DIR`; downloading EDICT2 and importing a `ta.db` run as jobs on the existing worker. The SPA gains a setup wizard, an `/admin` page, and a `ta.db` card on `/import`.

**Tech Stack:** FastAPI, SQLAlchemy 2 + Alembic, Postgres 16, httpx, pytest; React + TanStack Query + react-router, openapi-typescript, Playwright; Caddy 2.

**Spec:** `docs/superpowers/specs/2026-10-16-web-onboarding-design.md`

## Global Constraints

- Every router is built with `api_router()`; handlers `flush()`, never `commit()` (`tests/test_commit_boundary.py`).
- Key-addressed deletes are idempotent `204`; id-addressed resources `404` when missing.
- No Python module over ~400 lines.
- Migrations are autogenerated (`alembic revision --autogenerate`); every model is imported in `api/models/__init__.py`.
- Config is env only (`TA_` prefix, `api/config.py`), plus the `server_settings` rows this plan adds.
- Stylesheets use tokens only, never literal colours; reuse existing classes (`panel sec`, `sec-head`, `field`, `toolbar`, `primary`, `box ok`, `box danger`, `muted`, `hint`, `mono`, `error`, `saved-note`).
- Forms that show loaded values render only after the data arrives and are keyed by it (a late remount must not wipe typed input).
- After any API change: `cd web && npm run gen:api` (needs the venv active).
- Tests use `httpx.MockTransport`, never live endpoints. Jobs in tests: `jobs.run_next()` with `api.jobs.SessionLocal` swapped for the test session.
- E2E: each spec owns its seeded rows in `tools/e2e_seed.py`; seed names must not contain "E2E chapter".
- Password minimum 12 characters. Server-setting validation: URLs empty or http/https; `syosetu_delay_seconds` 1–60.
- Limits: EDICT2 128 MiB unpacked, 32 MiB downloaded, 64 MiB per dictionary upload; `ta.db` 1 GiB per upload; uploads older than 24 h are swept at startup.
- EDICT2 source: `https://www.edrdg.org/pub/Nihongo/edict2.gz`, redirects not followed; credit EDRDG (CC BY-SA 4.0).
- Commit messages end with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

## Review Focus

1. A second browser tab still on wizard step 1 after the admin exists → `409` with "Setup is already complete; sign in instead." and the SPA shows a sign-in link (Task 1 test, Task 9 UI).
2. An admin pastes a URL with surrounding spaces or no scheme (`localhost:11434`) → spaces are stripped; no scheme is a `422` naming the field, nothing saved (Task 2 test).
3. A user uploads some other SQLite database (not a desktop `ta.db`) → the dry-run job fails with "not a desktop ta.db" and the upload file is deleted (Task 7 test).
4. The server has no internet when the admin clicks Download → job fails with "could not reach edrdg.org", the old dictionary is kept (Task 5 test).
5. A `ta.db` over the cap → `413` and no partial file left behind (Task 7 test).

```python
"""Account creation shared by the setup wizard, the seed CLI and login's session start."""

from datetime import UTC, datetime, timedelta

from fastapi import Response
from sqlalchemy import func, select
from sqlalchemy.orm import Session

from api.config import get_settings
from api.deps import COOKIE_NAME
from api.models import Profile, User
from api.models import Session as SessionRow
from api.security import hash_password, new_token, token_hash

MIN_PASSWORD = 12
# Any constant works; it only has to be the same for every claim (pg_advisory_xact_lock key).
SETUP_LOCK = 0x7461_5345_5455


class AlreadySetUp(Exception):
    """A user exists, so the first-admin claim is closed."""


def create_admin(db: Session, email: str, password: str) -> User:
    email = email.strip().lower()
    if len(password) < MIN_PASSWORD:
        raise ValueError(f"password must be at least {MIN_PASSWORD} characters")
    if db.scalar(select(User).where(User.email == email)) is not None:
        raise ValueError(f"user {email} already exists")
    user = User(email=email, password_hash=hash_password(password), is_admin=True)
    db.add(user)
    db.flush()
    db.add(Profile(user_id=user.id, name="default", is_default=True))
    db.flush()
    return user


def needs_setup(db: Session) -> bool:
    return db.scalar(select(User.id).limit(1)) is None


def claim_first_admin(db: Session, email: str, password: str) -> User:
    """Create the first admin, or raise AlreadySetUp. The transaction-scoped advisory lock
    serialises concurrent claims: the second waits, then sees the first's user."""
    db.execute(select(func.pg_advisory_xact_lock(SETUP_LOCK)))
    if not needs_setup(db):
        raise AlreadySetUp
    return create_admin(db, email, password)


def start_session(db: Session, response: Response, user: User) -> None:
    settings = get_settings()
    token = new_token()
    db.add(SessionRow(
        token_hash=token_hash(token),
        user_id=user.id,
        expires_at=datetime.now(UTC) + timedelta(days=settings.session_ttl_days),
    ))
    db.flush()
    response.set_cookie(
        COOKIE_NAME, token, httponly=True, samesite="lax", secure=settings.cookie_secure,
        max_age=settings.session_ttl_days * 86400, path="/",
    )
```

```python
"""First-run setup (spec §3): open to anyone until the first account exists."""

from fastapi import Depends, HTTPException, Response, status
from sqlalchemy.orm import Session

from api.accounts import AlreadySetUp, claim_first_admin, needs_setup, start_session
from api.deps import db_session
from api.models import User
from api.routing import api_router
from api.schemas.auth import UserOut
from api.schemas.setup import SetupAdmin, SetupStatus

router = api_router(prefix="/setup", tags=["setup"])


@router.get("/status", response_model=SetupStatus)
def setup_status(db: Session = Depends(db_session)) -> SetupStatus:
    return SetupStatus(needs_setup=needs_setup(db))


@router.post("/admin", response_model=UserOut, status_code=status.HTTP_201_CREATED)
def claim_admin(body: SetupAdmin, response: Response, db: Session = Depends(db_session)) -> User:
    try:
        user = claim_first_admin(db, body.email, body.password)
    except AlreadySetUp:
        raise HTTPException(
            status.HTTP_409_CONFLICT, "Setup is already complete; sign in instead."
        ) from None
    except ValueError as exc:
        raise HTTPException(status.HTTP_422_UNPROCESSABLE_CONTENT, str(exc)) from exc
    start_session(db, response, user)
    return user
```

```python
"""Server-wide settings an admin edits in the browser (spec §4). Value = DB row, else the
env var of the same name (api/config.py), else its code default. No cache: one small query."""

from dataclasses import dataclass
from urllib.parse import urlsplit

from sqlalchemy import delete, func, select
from sqlalchemy.dialects.postgresql import insert
from sqlalchemy.orm import Session

from api.config import get_settings
from api.models import ServerSetting


def _url(key, v):
    if not isinstance(v, str):
        raise ValueError(f"{key} must be text")
    v = v.strip()
    parts = urlsplit(v)
    if v and (parts.scheme not in ("http", "https") or not parts.netloc):
        raise ValueError(f"{key} must be an http:// or https:// address")
    return v


def _bool(key, v):
    if not isinstance(v, bool):
        raise ValueError(f"{key} must be true or false")
    return v


def _delay(key, v):
    if isinstance(v, bool) or not isinstance(v, int | float) or not 1 <= v <= 60:
        raise ValueError(f"{key} must be a number of seconds from 1 to 60")
    return float(v)


VALIDATORS = {
    "ollama_url": _url,
    "libretranslate_url": _url,
    "mt_allow_private_urls": _bool,
    "wp_allow_private_urls": _bool,
    "syosetu_delay_seconds": _delay,
}
KEYS = tuple(VALIDATORS)


@dataclass(frozen=True)
class ServerPrefs:
    ollama_url: str
    libretranslate_url: str
    mt_allow_private_urls: bool
    wp_allow_private_urls: bool
    syosetu_delay_seconds: float


def validate(key: str, value):
    """The cleaned value; KeyError for an unknown key, ValueError for a bad value."""
    return VALIDATORS[key](key, value)


def _rows(db: Session) -> dict:
    return dict(db.execute(select(ServerSetting.key, ServerSetting.value)).all())


def effective(db: Session) -> ServerPrefs:
    rows, env = _rows(db), get_settings()
    return ServerPrefs(**{k: rows[k] if k in rows else getattr(env, k) for k in KEYS})


def _entry(key: str, rows: dict) -> dict:
    env = get_settings()
    if key in rows:
        return {"key": key, "value": rows[key], "source": "db"}
    # pydantic-settings records env-sourced fields in model_fields_set
    source = "env" if key in env.model_fields_set else "default"
    return {"key": key, "value": getattr(env, key), "source": source}


def listing(db: Session) -> list[dict]:
    rows = _rows(db)
    return [_entry(k, rows) for k in KEYS]


def put(db: Session, key: str, value, user_id: int) -> dict:
    value = validate(key, value)
    stmt = insert(ServerSetting).values(key=key, value=value, updated_by=user_id)
    db.execute(stmt.on_conflict_do_update(
        index_elements=[ServerSetting.key],
        set_={"value": stmt.excluded.value, "updated_by": user_id, "updated_at": func.now()},
    ))
    db.flush()
    return _entry(key, {key: value})


def reset(db: Session, key: str) -> None:
    if key not in VALIDATORS:
        raise KeyError(key)
    db.execute(delete(ServerSetting).where(ServerSetting.key == key))
    db.flush()
```

```python
"""Admin-only server settings (spec §4.4)."""

from fastapi import Depends, HTTPException, status
from sqlalchemy.orm import Session

from api import server_settings
from api.deps import db_session, require_admin
from api.models import User
from api.routing import api_router
from api.schemas.admin import ServerSettingOut, SettingValue

router = api_router(prefix="/admin", tags=["admin"])


def _unknown(key: str) -> HTTPException:
    return HTTPException(status.HTTP_404_NOT_FOUND, f"no server setting named {key!r}")


@router.get("/settings", response_model=list[ServerSettingOut])
def list_settings(_: User = Depends(require_admin), db: Session = Depends(db_session)) -> list[dict]:
    return server_settings.listing(db)


@router.put("/settings/{key}", response_model=ServerSettingOut)
def put_setting(key: str, body: SettingValue, admin: User = Depends(require_admin),
                db: Session = Depends(db_session)) -> dict:
    try:
        return server_settings.put(db, key, body.value, admin.id)
    except KeyError:
        raise _unknown(key) from None
    except ValueError as exc:
        raise HTTPException(status.HTTP_422_UNPROCESSABLE_CONTENT, str(exc)) from exc


@router.delete("/settings/{key}", status_code=status.HTTP_204_NO_CONTENT)
def reset_setting(key: str, _: User = Depends(require_admin),
                  db: Session = Depends(db_session)) -> None:
```

```python
"""EDICT2 and Conjugations files kept under TA_DATA_DIR (spec §5). An explicit
TA_EDICT_PATH / TA_CONJUGATIONS_PATH wins, so env-configured installs are unchanged."""

import os
import tempfile
import zlib
from datetime import UTC, datetime
from pathlib import Path
from typing import Literal

from api.config import get_settings

Kind = Literal["edict", "conjugations"]
FILENAMES = {"edict": "edict2", "conjugations": "conjugations.txt"}
ENV_FIELDS = {"edict": "edict_path", "conjugations": "conjugations_path"}
MIN_EDICT_ENTRIES = 1000  # the real file has ~190k; this only rejects the wrong file
MAX_EDICT_BYTES = 128 * 1024 * 1024
MAX_UPLOAD = 64 * 1024 * 1024


def dict_dir() -> Path:
    return Path(get_settings().data_dir) / "dict"


def env_path(kind: Kind) -> str:
    return getattr(get_settings(), ENV_FIELDS[kind])


def resolve(kind: Kind) -> tuple[str, str | None]:
    if p := env_path(kind):
        return p, "env"
    p = dict_dir() / FILENAMES[kind]
    return (str(p), "data_dir") if p.is_file() else ("", None)


def path(kind: Kind) -> str:
    return resolve(kind)[0]


def gunzip_capped(data: bytes) -> bytes:
    """Plain bytes pass through; gzip is unpacked, refusing more than MAX_EDICT_BYTES."""
    if not data.startswith(b"\x1f\x8b"):
        out, more = data, False
    else:
        d = zlib.decompressobj(16 + zlib.MAX_WBITS)
        try:
            out = d.decompress(data, MAX_EDICT_BYTES + 1)
        except zlib.error as exc:
            raise ValueError(f"the file is not valid gzip: {exc}") from exc
        more = bool(d.unconsumed_tail)
    if more or len(out) > MAX_EDICT_BYTES:
        raise ValueError(f"the dictionary is larger than {MAX_EDICT_BYTES >> 20} MiB unpacked")
    return out


def _count(kind: Kind, p: Path) -> int:
    from api.mt import jparser  # jparser resolves its paths through this module

    if kind == "edict":
        n = len(jparser._read_index(str(p)) or {})
        if n < MIN_EDICT_ENTRIES:
            raise ValueError("this does not look like EDICT2: too few dictionary entries")
        return n
    try:
        n = len(jparser._load_conj_rules(str(p)))
    except (ValueError, KeyError, TypeError) as exc:  # json and unicode errors are ValueError
        raise ValueError(f"this is not a Conjugations file: {exc}") from exc
    if n == 0:
        raise ValueError("this is not a Conjugations file: it has no rules")
    return n


def install(kind: Kind, data: bytes) -> int:
    """Validate, then atomically replace the data-dir file. Returns the entry/rule count."""
    d = dict_dir()
    d.mkdir(parents=True, exist_ok=True)
    fd, tmp = tempfile.mkstemp(dir=d, prefix=FILENAMES[kind] + ".", suffix=".tmp")
    try:
        with os.fdopen(fd, "wb") as f:
            f.write(data)
        n = _count(kind, Path(tmp))
        os.replace(tmp, d / FILENAMES[kind])
    except BaseException:
        Path(tmp).unlink(missing_ok=True)
        raise
    return n


def status(kind: Kind) -> dict:
    from api.mt import jparser

    p, source = resolve(kind)
    try:
        st = os.stat(p) if p else None
    except OSError:
        st = None
    if st is None:
        return {"present": False, "source": source, "size": None, "updated_at": None,
                "entries": None}
    entries = (len(jparser.load_index(p) or {}) if kind == "edict"
               else len(jparser.load_rules(p)))
    return {"present": True, "source": source, "size": st.st_size,
            "updated_at": datetime.fromtimestamp(st.st_mtime, UTC), "entries": entries}


def remove(kind: Kind) -> None:
    (dict_dir() / FILENAMES[kind]).unlink(missing_ok=True)
```

```python
        with http_client() as c, c.stream("GET", EDICT_URL) as r:
            if r.status_code != 200:
                raise JobError(f"edrdg.org answered {r.status_code}")
            for chunk in r.iter_bytes():
                buf += chunk
                if len(buf) > MAX_DOWNLOAD:
                    raise JobError(f"the download is larger than {MAX_DOWNLOAD >> 20} MiB")
    except httpx.HTTPError as exc:
        raise JobError(f"could not reach edrdg.org: {exc}") from exc
    ctx.progress(1, 2, "Installing")
    try:
        entries = dictionary.install("edict", dictionary.gunzip_capped(bytes(buf)))
    except ValueError as exc:
        raise JobError(str(exc)) from exc
    return {"entries": entries}
```

```python
"""Uploaded desktop ta.db files waiting for their import (spec §6). Streamed to disk:
a library with images can be hundreds of MB."""

import time
from datetime import timedelta
from pathlib import Path
from uuid import UUID, uuid4

from fastapi import HTTPException, UploadFile, status

from api.config import get_settings

MAX_TADB_BYTES = 1 << 30
MAGIC = b"SQLite format 3\x00"


def _root() -> Path:
    return Path(get_settings().data_dir) / "uploads"


def path_for(user_id: int, upload_id: str) -> Path:
    """The file for this user's upload id (it may not exist). ValueError for a malformed id."""
    return _root() / str(user_id) / f"{UUID(upload_id).hex}.db"


def save(user_id: int, upload: UploadFile) -> str:
    upload_id = uuid4().hex
    dest = path_for(user_id, upload_id)
    dest.parent.mkdir(parents=True, exist_ok=True)
    total = 0
    try:
        with dest.open("wb") as out:
            while chunk := upload.file.read(1 << 20):
                total += len(chunk)
                if total > MAX_TADB_BYTES:
                    raise HTTPException(status.HTTP_413_CONTENT_TOO_LARGE,
                                        f"upload exceeds {MAX_TADB_BYTES} bytes")
                out.write(chunk)
        with dest.open("rb") as f:
            if f.read(len(MAGIC)) != MAGIC:
                raise HTTPException(status.HTTP_422_UNPROCESSABLE_CONTENT,
                                    "This is not a SQLite database; choose the desktop ta.db.")
    except BaseException:
        dest.unlink(missing_ok=True)
        raise
    return upload_id


def sweep(max_age: timedelta = timedelta(hours=24)) -> int:
    """Delete uploads older than max_age (abandoned dry runs). Returns how many."""
    cutoff, n = time.time() - max_age.total_seconds(), 0
    for p in _root().glob("*/*.db"):
        try:
            if p.stat().st_mtime < cutoff:
                p.unlink()
                n += 1
        except FileNotFoundError:
            pass
    return n
```

```python
"""The browser ta.db import (spec §6.4): a dry run, then the real run, on the job worker."""

import sqlite3

from api import tadb_uploads
from api.core.tadb_import import Refused, import_tadb
from api.jobs import JobContext, JobError, handler


def _result(dry: bool, report=None, refused: str | None = None) -> dict:
    return {
        "dry_run": dry, "refused": refused,
        "written": dict(report.written) if report else {},
        "skipped": dict(report.skipped) if report else {},
        "already_present": dict(report.already_present) if report else {},
        "messages": list(report.messages) if report else [],
    }


@handler("tadb_import")
def run_import(ctx: JobContext) -> dict:
    p, user_id = ctx.params, ctx.job.user_id
    dry = bool(p["dry_run"])
    path = tadb_uploads.path_for(user_id, p["upload_id"])
    keep = False  # only a finished dry run keeps the file for the real run
    try:
        if not path.is_file():
            raise JobError("The uploaded file is gone; upload it again.")
        ctx.progress(0, 1, "Checking the library" if dry else "Importing the library")
        try:
            report = import_tadb(ctx.db, path, user_id, dry_run=dry, force=bool(p["force"]))
        except Refused as exc:
            if not dry:
                raise JobError(str(exc)) from exc
            keep = True
            return _result(dry, refused=str(exc))
        except RuntimeError as exc:  # schema checks: "... not a desktop ta.db"
            raise JobError(str(exc)) from exc
        except sqlite3.DatabaseError as exc:
            raise JobError(f"not a desktop ta.db: {exc}") from exc
        keep = dry
        return _result(dry, report)
    finally:
        if not keep:
            path.unlink(missing_ok=True)
```

```tsx
type Entry = Schemas["ServerSettingOut"];

  { key: "ollama_url", label: "Ollama URL", kind: "url", hint: "e.g. http://192.168.1.20:11434" },
  { key: "libretranslate_url", label: "LibreTranslate URL", kind: "url", hint: "Leave empty if you do not run one." },

] as const;
const BADGE: Record<Entry["source"], string> = { db: "", env: "from env", default: "default" };

function read(f: FormData, kind: string, key: string): unknown {
  if (kind === "bool") return f.get(key) === "on";
  if (kind === "number") return Number(f.get(key));
  return String(f.get(key) ?? "").trim();
}

export default function ServerSettingsForm({ onSaved }: { onSaved?: () => void }) {
  const queryClient = useQueryClient();
  const [note, setNote] = useState("");
  const q = useQuery({ queryKey: ["admin", "settings"], queryFn: () => call(api.GET("/admin/settings")) });
  const refresh = (msg: string) => { setNote(msg); return queryClient.invalidateQueries({ queryKey: ["admin", "settings"] }); };
  const save = useMutation({
    mutationFn: async (changed: [string, unknown][]) => {
      for (const [key, value] of changed) await call(api.PUT("/admin/settings/{key}", { params: { path: { key } }, body: { value } }));
    },
    onSuccess: () => { void refresh("Saved"); onSaved?.(); },
  });
  const reset = useMutation({
    mutationFn: (key: string) => call(api.DELETE("/admin/settings/{key}", { params: { path: { key } } })),
    onSuccess: () => refresh("Reset"),
  });
  if (q.isPending) return <p className="muted">Loading…</p>;
  if (q.isError) return <p className="error">{q.error.message}</p>;
  const byKey = Object.fromEntries(q.data.map((e) => [e.key, e]));
  const submit = (e: FormEvent<HTMLFormElement>) => {
    e.preventDefault();
    const f = new FormData(e.currentTarget);
    // Only changed values: saving an untouched field would pin it and hide later env changes.
    const changed = FIELDS.map((d) => [d.key, read(f, d.kind, d.key)] as [string, unknown])
      .filter(([key, value]) => value !== byKey[key].value);
    if (changed.length) save.mutate(changed);
    else onSaved?.();
  };
  const error = save.error ?? reset.error;
  return (
    <form key={JSON.stringify(q.data)} onSubmit={submit} aria-label="Server settings">
      {FIELDS.map((d) => {
        const e = byKey[d.key];
        const badge = BADGE[e.source];
        return (
          <div key={d.key} className="field">
            {d.kind === "bool" ? (
              <label><input type="checkbox" name={d.key} defaultChecked={Boolean(e.value)} /> {d.label}</label>
            ) : (
              <label><span>{d.label}</span>
                <input name={d.key} className={d.kind === "url" ? "mono" : undefined}
                  {...(d.kind === "number" ? { type: "number", min: 1, max: 60, step: "any" } : { type: "text" })}
                  defaultValue={String(e.value)} /></label>
            )}
            <p className="hint">{d.hint}{badge && <> · <span className="muted">{badge}</span></>}

          </div>
        );
      })}
      {error && <p className="error" role="alert">{error.message}</p>}
      <div className="toolbar">
        <button type="submit" className="primary" disabled={save.isPending}>Save</button>
        {note && <span role="status" className="saved-note">{note}</span>}
      </div>
    </form>
  );
```
